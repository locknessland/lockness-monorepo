# RootLayout

Documentation for the RootLayout component.

## Installation

```bash
deno run -A jsr:@lockness/ui add rootlayout
```

## Usage

```tsx
import { RootLayout } from '@lockness/ui/components'

export const HomePage = () => (
    <RootLayout title='Home'>
        <div class='container mx-auto p-4'>
            <h1>Welcome to my app</h1>
        </div>
    </RootLayout>
)
```
